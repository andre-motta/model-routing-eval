import time
from collections import OrderedDict


class LRUCache:
    def __init__(self, capacity, ttl=None, clock=time.monotonic):
        if capacity <= 0:
            raise ValueError("capacity must be positive")
        self.capacity = capacity
        self.ttl = ttl
        self.clock = clock
        self._items = OrderedDict()
        self._stats = {"hits": 0, "misses": 0, "evictions": 0, "expirations": 0}

    def _expired(self, key, now=None):
        entry = self._items.get(key)
        if entry is None:
            return False
        expires_at = entry[1]
        if expires_at is not None and (self.clock() if now is None else now) >= expires_at:
            del self._items[key]
            self._stats["expirations"] += 1
            return True
        return False

    def get(self, key, default=None):
        if self._expired(key):
            self._stats["misses"] += 1
            return default
        try:
            value, expires_at = self._items.pop(key)
        except KeyError:
            self._stats["misses"] += 1
            return default
        self._items[key] = (value, expires_at)
        self._stats["hits"] += 1
        return value

    def put(self, key, value):
        now = self.clock()
        # Replacing an already expired key starts a fresh lifetime, while recording
        # the expiration that was observed by this operation.
        self._expired(key, now)
        expires_at = None if self.ttl is None else now + self.ttl
        self._items.pop(key, None)
        self._items[key] = (value, expires_at)
        if len(self._items) > self.capacity:
            self._items.popitem(last=False)
            self._stats["evictions"] += 1

    def delete(self, key):
        if self._expired(key):
            return False
        return self._items.pop(key, None) is not None

    def stats(self):
        return self._stats.copy()

    def __len__(self):
        now = self.clock()
        expired = [key for key, (_, expires_at) in self._items.items()
                   if expires_at is not None and now >= expires_at]
        for key in expired:
            del self._items[key]
            self._stats["expirations"] += 1
        return len(self._items)

    def __contains__(self, key):
        if self._expired(key):
            return False
        return key in self._items

def test_lru_eviction_and_recency():

    cache.put("c", 3)
    assert "b" not in cache

def test_update_moves_entry_to_most_recent():

    cache.put("a", 3)

    assert cache.get("a") == 3
    assert "b" not in cache


def test_ttl_expiration_and_operations():
